#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <UniversalTelegramBot.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Wire.h>

// --- CONFIG WI-FI ---
const char* ssid = "AksangKel2";
const char* password = "qqqqqqqq";

// --- CONFIG MQTT BROKER PPTIK ---
const char* mqtt_server = "rmq230.pptik.id";
const int mqtt_port = 1883;
const char* mqtt_user = "/prd:prd26-kel11";
const char* mqtt_pass = "Smart123System";
const char* mqtt_topic = "Trigger";

// --- CONFIG TELEGRAM NOTIFIKASI ---
#define BOTtoken "8609376337:AAFM-5u5YmcfEKwlL6zdaxexBgknNYo7IxI"
#define CHAT_ID "7416287973"

// --- PIN HARDWARE ---
#define BUZZER_PIN 14
#define PIR_PIN    12

// --- MPU6050 CONFIG ---
Adafruit_MPU6050 mpu;
// Ambang batas getaran/guncangan (m/s^2). Default gravitas bumi ~9.8 m/s^2.
// Nilai 14.0 berarti mendeteksi akselerasi > 14 m/s^2 (getaran/ketukan cukup kuat).
const float MPU_THRESHOLD = 14.0; 

// Network Clients
WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient);

WiFiClientSecure telegramClient;
UniversalTelegramBot bot(BOTtoken, telegramClient);

// --- TIMING & STATE VARIABLES ---
unsigned long pirCooldown = 0;
unsigned long mpuCooldown = 0;
unsigned long buzzerTurnOffTime = 0;

bool triggerPending = false;
bool buzzerActive = false;

// Fungsi Mengirim Pesan Telegram
void sendTelegramMsg(String message) {
  telegramClient.setInsecure(); // Mengabaikan verifikasi sertifikat SSL
  bot.sendMessage(CHAT_ID, message, "");
}

// Callback ketika ada pesan MQTT masuk dari Broker
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  Serial.print("[MQTT Received] Topic: ");
  Serial.print(topic);
  Serial.print(" | Message: ");
  Serial.println(message);

  if (message == "TRIGGER") {
    triggerPending = true;
  }
}

// Fungsi Reconnect MQTT jika terputus
void reconnectMQTT() {
  while (!mqttClient.connected()) {
    Serial.print("Mencoba konek ke Broker PPTIK (rmq230.pptik.id)...");
    
    // Generate Random Client ID agar tidak konflik
    String clientId = "ESP32Aktuator-";
    clientId += String(random(0xffff), HEX);

    // Konek dengan autentikasi Username & Password
    if (mqttClient.connect(clientId.c_str(), mqtt_user, mqtt_pass)) {
      Serial.println(" Terhubung!");
      mqttClient.subscribe(mqtt_topic);
      Serial.print("Subscribed ke topic: ");
      Serial.println(mqtt_topic);
    } else {
      Serial.print(" Gagal, rc=");
      Serial.print(mqttClient.state());
      Serial.println(" Mencoba lagi dalam 3 detik...");
      delay(3000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(PIR_PIN, INPUT);
  digitalWrite(BUZZER_PIN, LOW);

  // Inisialisasi MPU6050
  Wire.begin(); // Default SDA=21, SCL=22
  if (!mpu.begin()) {
    Serial.println("❌ Gagal menemukan sensor MPU6050! Periksa kabel I2C Anda.");
  } else {
    Serial.println("✅ Sensor MPU6050 berhasil diinisialisasi.");
    mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
    mpu.setGyroRange(MPU6050_RANGE_500_DEG);
    mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
  }

  // Konek Wi-Fi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");
  Serial.print("ESP32 Aktuator IP: ");
  Serial.println(WiFi.localIP());

  // Setup Server MQTT & Callback
  mqttClient.setServer(mqtt_server, mqtt_port);
  mqttClient.setCallback(mqttCallback);
}

void loop() {
  // Menjaga koneksi MQTT tetap terhubung
  if (!mqttClient.connected()) {
    reconnectMQTT();
  }
  mqttClient.loop();

  unsigned long currentMillis = millis();

  // --- 1. PROSES TRIGGER DARI MQTT (AI DETEKSI) ---
  if (triggerPending) {
    triggerPending = false;
    
    // Nyalakan buzzer selama 500ms
    digitalWrite(BUZZER_PIN, HIGH);
    buzzerActive = true;
    buzzerTurnOffTime = currentMillis + 500;
    Serial.println("Buzzer Hidup (Sampah AI)");

    // Kirim notifikasi Telegram
    Serial.println("Telegram masuk (Sampah AI)");
    sendTelegramMsg("⚠️ [AI CCTV] Sampah terdeteksi oleh kamera!");
  }

  // --- 2. CEK SENSOR MPU6050 (GETARAN/GUNCANGAN - COOLDOWN 5 DETIK) ---
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  // Hitung magnitudo percepatan total: sqrt(x^2 + y^2 + z^2)
  float totalAccel = sqrt(a.acceleration.x * a.acceleration.x +
                          a.acceleration.y * a.acceleration.y +
                          a.acceleration.z * a.acceleration.z);

  if (totalAccel > MPU_THRESHOLD && (currentMillis - mpuCooldown > 5000)) {
    digitalWrite(BUZZER_PIN, HIGH);
    buzzerActive = true;
    buzzerTurnOffTime = currentMillis + 400; // Beep 400ms
    Serial.println("Buzzer hidup (MPU6050 Getaran)");

    Serial.println("Telegram masuk (MPU6050)");
    sendTelegramMsg("📳 [MPU6050] Terdeteksi guncangan/getaran pada tempat sampah!");
    mpuCooldown = currentMillis;
  }

  // --- 3. CEK SENSOR PIR (GERAKAN - COOLDOWN 5 DETIK) ---
  if (digitalRead(PIR_PIN) == HIGH && (currentMillis - pirCooldown > 5000)) {
    digitalWrite(BUZZER_PIN, HIGH);
    buzzerActive = true;
    buzzerTurnOffTime = currentMillis + 200; // Beep pendek 200ms
    Serial.println("Buzzer hidup (PIR)");

    Serial.println("Telegram masuk (PIR)");
    sendTelegramMsg("🚶 Gerakan terdeteksi oleh Sensor PIR!");
    pirCooldown = currentMillis;
  }

  // --- 4. MEMATIKAN BUZZER TANPA BLOCKING DELAY ---
  if (buzzerActive && currentMillis >= buzzerTurnOffTime) {
    digitalWrite(BUZZER_PIN, LOW);
    Serial.println("Buzzer Mati");
    buzzerActive = false;
  }

  delay(50); // Delay kecil agar pembacaan MPU6050 & loop lebih halus
}
